Part 1 — k-means from scratch (Code S6) and the bad-start trap

Goal: Run the course guide's Code S6, then watch inertia fall iteration by iteration and see why scikit-learn uses k-means++ and several starts.

k-means is two lines in a loop: assign every point to the nearest centroid (argmin of squared distances) and update every centroid to the mean of its members. It always stops, but only at a local best.

def kmeans(P, k, seed=0, iters=20, verbose=True):
    r = np.random.default_rng(seed)
    C = P[r.choice(len(P), k, replace=False)]
    for it in range(iters):
        d2 = ((P[:, None] - C) ** 2).sum(axis=2)          # squared distance of every point to every centroid
        lab = d2.argmin(axis=1)                             # assign: nearest centroid
        inertia = d2[np.arange(len(P)), lab].sum()          # WCSS for this assignment
        newC = np.array([P[lab == j].mean(axis=0) for j in range(k)])   # update: mean of members
        if verbose: print(f"iteration {it + 1}: inertia {inertia:9.2f}")
        if np.allclose(newC, C): break
        C = newC
    return C, lab, inertia

import time
from sklearn.base import BaseEstimator, TransformerMixin
from sklearn.pipeline import make_pipeline, make_union
from sklearn.preprocessing import FunctionTransformer
from sklearn.compose import ColumnTransformer
from sklearn.linear_model import LogisticRegression
from sklearn.svm import SVC
from sklearn.model_selection import cross_val_score

class ClusterOneHot(BaseEstimator, TransformerMixin):
    """Fits k-means on the training fold; outputs a one-hot column per cluster."""
    def __init__(self, k=4): self.k = k
    def fit(self, X, y=None): self.km_ = KMeans(n_clusters=self.k, n_init=10, random_state=42).fit(X); return self
    def transform(self, X): return np.eye(self.k)[self.km_.predict(X)]
