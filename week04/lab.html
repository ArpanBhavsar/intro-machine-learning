Part 1 — Scam Shield: Naive Bayes from scratch and in scikit-learn

Goal: Run the course guide's Code S4, build a word-counting Naive Bayes spam filter yourself, then compare it with CountVectorizer and TfidfVectorizer + MultinomialNB.

Training Naive Bayes is counting: words per class plus the class prior. Prediction adds log probabilities (to avoid underflow) with Laplace smoothing (+1) so a word unseen in one class does not veto the whole message. Words never seen in training are skipped.

# --- Code S4 (course guide): the same Naive Bayes idea on six tiny messages; expected output: S4 spam ham
from collections import Counter
import math
train = [("win free cash now", "spam"), ("free prize claim now", "spam"),
         ("are we meeting for lunch", "ham"), ("see you at class tomorrow", "ham"),
         ("claim your free gift", "spam"), ("lunch tomorrow at noon", "ham")]
counts = {"spam": Counter(), "ham": Counter()}; docs = Counter()
for text, label in train:
    counts[label].update(text.split()); docs[label] += 1
vocab = set(w for c in counts.values() for w in c)
def nb_predict(text):
    scores = {}
    for label in counts:
        total = sum(counts[label].values())
        score = math.log(docs[label] / sum(docs.values()))              # prior
        for word in text.split():                                          # likelihoods (+1 smoothing)
            score += math.log((counts[label][word] + 1) / (total + len(vocab)))
        scores[label] = score
    return max(scores, key=scores.get)
print("S4", nb_predict("free cash prize"), nb_predict("lunch at class"))

import re
from collections import Counter

tokens = lambda s: re.findall(r"[a-z0-9']+", s.lower())     # lower-case words and numbers

# 1) COUNT: how often each word appears in spam and in ham (training messages only)
spam_c, ham_c = Counter(), Counter()
for text, lab in zip(s_train, l_train):
    (spam_c if lab == 1 else ham_c).update(tokens(text))
vocab = set(spam_c) | set(ham_c); V = len(vocab)
n_spam, n_ham = sum(spam_c.values()), sum(ham_c.values())
prior_spam = l_train.mean()

# 2) SCORE: log P(class) + sum of log P(word | class), with Laplace smoothing (+1)
def score(text, c, n, prior):
    return np.log(prior) + sum(np.log((c[w] + 1) / (n + V)) for w in tokens(text) if w in vocab)

def predict(text):
    return int(score(text, spam_c, n_spam, prior_spam) > score(text, ham_c, n_ham, 1 - prior_spam))

pred = np.array([predict(t) for t in s_test])
tp = ((pred == 1) & (l_test == 1)).sum(); fp = ((pred == 1) & (l_test == 0)).sum(); fn = ((pred == 0) & (l_test == 1)).sum()
print("vocabulary:", V, " spam words:", n_spam, " ham words:", n_ham, " prior P(spam) = %.4f" % prior_spam)
print("from scratch: accuracy %.4f  precision %.4f  recall %.4f" % ((pred == l_test).mean(), tp / (tp + fp), tp / (tp + fn)))

# --- margin and hinge loss by hand for two candidate lines (Math toolkit slide)
P5 = np.array([[1, 1], [2, 1.5], [3, 3], [2.6, 2.6], [4, 3]]); y5 = np.array([-1, -1, 1, 1, 1])
for name, w, b in [("narrow", np.array([1.0, 1.0]), -4.0), ("wide", np.array([0.5, 0.5]), -2.0)]:
    f = P5 @ w + b                                    # signed score of each point
    hinge = np.maximum(0, 1 - y5 * f)                 # 0 if outside the street on the right side
    width = 2 / np.linalg.norm(w)                     # street width = 2 / ||w||
    print("%-6s f=%s hinge=%s width=%.3f" % (name, f.round(2), hinge.round(2), width), end="  ")
    print("objective C=0.1: %.3f  C=1: %.3f  C=10: %.3f" % tuple(0.5 * w @ w + C * hinge.sum() for C in (0.1, 1, 10)))
# what scikit-learn's linear SVM chooses on the same 5 points
for C in (0.1, 10):
    lin = SVC(kernel="linear", C=C).fit(P5, y5); w = lin.coef_[0]
    print("SVC(linear, C=%-4s) w=%s b=%.3f width=%.3f support vectors=%d" % (C, w.round(3), lin.intercept_[0], 2 / np.linalg.norm(w), lin.n_support_.sum()))

    ax.plot(fpr, tpr, label="%s (AUC %.3f)" % (name, roc_auc_score(y_test, score)))
ax.plot([0, 1], [0, 1], "k--", lw=1); ax.set_xlabel("false positive rate"); ax.set_ylabel("true positive rate (recall)"); ax.legend(fontsize=8)
fig.tight_layout(); fig.savefig("lab-assets/roc_top3.png", dpi=110); print("saved lab-assets/roc_top3.png")
