Solution
import numpy as np
rng = np.random.default_rng(0)
Xh = rng.uniform(0, 10, 100)                 # hours studied
yh = 3.0 * Xh + 5 + rng.normal(0, 2, 100)    # exam score (true w=3, b=5)

def train(lr, steps=2000):
    w, b, history = 0.0, 0.0, []
    for step in range(steps):
        y_hat = w * Xh + b
        error = y_hat - yh
        loss = (error ** 2).mean()           # MSE
        history.append(loss)
        dw = 2 * (error * Xh).mean()         # dLoss/dw
        db = 2 * error.mean()                # dLoss/db
        w -= lr * dw
        b -= lr * db
    return w, b, history

w, b, hist = train(0.01)
print(f"S1 w={w:.2f} b={b:.2f} loss={hist[-1]:.2f}")

with np.errstate(over="ignore", invalid="ignore"):
    for lr in [0.0001, 0.01, 0.1]:
        w_, b_, h = train(lr)
        print(f"lr={lr:<7} loss after 10 steps={h[10]:.4g}  after 2000 steps={h[-1]:.4g}  w={w_:.3g}")

import matplotlib.pyplot as plt
with np.errstate(over="ignore", invalid="ignore"):

Solution
# Math check: the gradient formula equals the "nudge" slope, and gradient descent equals the exact answer
L = lambda w, b: ((w * Xh + b - yh) ** 2).mean()                  # MSE for any w, b
w0, b0, h = 0.0, 0.0, 1e-5
err = w0 * Xh + b0 - yh
dw_formula = 2 * (err * Xh).mean()                               # (2/n) Σ (ŷ - y) x
dw_nudge = (L(w0 + h, b0) - L(w0 - h, b0)) / (2 * h)              # rise over run
print(f"slope at w=0: formula {dw_formula:.4f}   nudge {dw_nudge:.4f}")
w1 = w0 - 0.01 * dw_formula                                      # one update step
print(f"one step (lr 0.01): w = 0 - 0.01 x ({dw_formula:.2f}) = {w1:.4f};  loss {L(w0, b0):.2f} -> {L(w1, b0):.2f}")
w_ls, b_ls = np.polyfit(Xh, yh, 1)                               # exact least-squares line
print(f"gradient descent w={w:.4f} b={b:.4f}   exact least squares w={w_ls:.4f} b={b_ls:.4f}")

Both slopes are −288.1083: the formula is just an exact, fast version of rise over run. The negative slope means downhill is towards larger w, so one step with lr = 0.01 moves w from 0 to 2.881 and the loss falls from 538.29 to 34.02. After 2,000 steps gradient descent gives w = 2.9739 and b = 4.9846, matching the exact least-squares line (2.9739, 4.9851) to three decimals; b is still creeping the last 0.0005.

def audit(data, target):
    """Quick data-quality report for any DataFrame (use it on your team dataset)."""
    print("Shape:", data.shape)
    print("Duplicate rows:", data.duplicated().sum())
    gaps = data.isna().sum()
    print("Columns with missing values:", gaps[gaps > 0].to_dict() or "none")
    kinds = {c: ("binary" if data[c].nunique() == 2 else "numeric" if pd.api.types.is_numeric_dtype(data[c]) else "categorical")
             for c in data.columns}
    print("Column kinds:", pd.Series(kinds).value_counts().to_dict())
    print(f"Target '{target}':")
    print(data[target].describe().round(2).to_string() if kinds[target] == "numeric" else data[target].value_counts(normalize=True).round(3).to_string())
